💾 จัดการและสำรองข้อมูลระบบ (Data Backup & Sync)

เนื่องจากระบบทำงานบนบราวเซอร์ภายในเครื่องของคุณ (Local Browser Storage) เพื่อความปลอดภัยสูงสุดและป้องกันข้อมูลสูญหาย คุณสามารถทำการสำรองไฟล์เก็บเข้าแฟลชไดรฟ์ หรืออัปโหลดเก็บขึ้นระบบคลาวด์ เช่น Google Drive ได้ตลอดเวลาครับ

1. ส่งออกข้อมูลสำรอง (Export Backup File)

ดาวน์โหลดไฟล์ฐานข้อมูลระบบทั้งหมด (รวมถึงรายชื่อคนงานต่างด้าว 3,000 คน นายจ้าง 300 ราย ข้อมูลสั่งงาน และยอดบัญชีธนาคารทั้งหมด) เป็นไฟล์เดียว เพื่อนำไปเซฟเก็บไว้ใน Google Drive, OneDrive หรือโฟลเดอร์ในเครื่องคอมพิวเตอร์ของคุณ

2. นำเข้าข้อมูลระบบจากภายนอก (Import Backup File)

เลือกไฟล์สำรองข้อมูลที่คุณเคยบันทึกไว้ (ไฟล์ .json) จากในเครื่องคอมพิวเตอร์ หรือดึงลงมาจาก Google Drive เพื่อกู้คืนฐานข้อมูลทั้งหมดกลับมาทำงานในเบราว์เซอร์นี้ทันที

ยังไม่ได้เลือกไฟล์

3. เชื่อมต่อ Google Sheets (Cloud Sync)

คุณสามารถเชื่อมโยงระบบนี้เข้ากับ **Google Sheets** ของคุณโดยตรง เมื่อมีการเพิ่มคนงานต่างด้าว นายจ้าง หรือใบแจ้งงานในโปรแกรม ระบบจะส่งข้อมูลไปบันทึกและเพิ่มแถวเข้าไปใน Google Sheets แบบ Real-time ทันที!

🛠️ วิธีการติดตั้งฝั่ง Google Sheets (ใช้เวลา 2 นาที):
1. เปิด Google Sheet ที่คุณต้องการใช้งานขึ้นมา ➡️ ไปที่เมนู **"ส่วนขยาย" (Extensions) ➡️ "Apps Script"**
2. ก๊อปปี้โค้ดโปรแกรมฝั่ง Server ด้านล่างนี้ไปวางลบอันเดิมออกทั้งหมด
3. กดปุ่ม **"การทำให้ใช้งานได้" (Deploy) ➡️ "การทำให้ใช้งานได้ใหม่" (New Deployment)**
4. เลือกประเภทเป็น **"เว็บแอป" (Web App)** ➡️ ตั้งค่า *ผู้ที่มีสิทธิ์เข้าถึง* เป็น **"ทุกคน" (Anyone)** ➡️ กด Deploy
5. ก๊อปปี้ **URL เว็บแอป** ที่ได้มาวางในช่องด้านบนนี้ แล้วกดทดสอบการเชื่อมต่อได้เลยครับ!
📋 คลิกเพื่อดูโค้ดสำหรับใส่ใน Apps Script 
// โค้ดสำหรับ Google Apps Script
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var json = JSON.parse(e.postData.contents);
    var action = json.action;
    var data = json.data;
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (action === "test") {
      return ContentService.createTextOutput(JSON.stringify({status: "success", message: "เชื่อมต่อสำเร็จ!"}))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader("Access-Control-Allow-Origin", "*");
    }
    
    var sheet = ss.getSheetByName(action) || ss.insertSheet(action);
    
    // ตั้งค่าหัวตารางหากเป็นชีทใหม่
    if (sheet.getLastRow() === 0) {
      if (action === "Customers") {
        sheet.appendRow(["TaxID", "CompanyName", "BusinessType", "Coordinator", "Phone", "UpdatedDate"]);
      } else if (action === "Workers") {
        sheet.appendRow(["WorkerUID", "Name", "Nationality", "PassportNo", "PermitNo", "PermitExpiry", "Employer", "UpdatedDate"]);
      } else if (action === "Jobs") {
        sheet.appendRow(["JobID", "CustomerName", "WorkerName", "JobType", "Fee", "Status", "Notes", "UpdatedDate"]);
      }
    }
    
    // บันทึกแถวข้อมูลใหม่
    if (action === "Customers") {
      sheet.appendRow([data.taxId, data.companyName, data.businessType, data.coordinator, data.phone, new Date()]);
    } else if (action === "Workers") {
      sheet.appendRow([data.workerUid, data.firstName + " " + data.lastName, data.nationality, data.passportNo, data.permitNo, data.permitExpiry, data.employerName, new Date()]);
    } else if (action === "Jobs") {
      sheet.appendRow([data.id, data.customerName, data.workerName, data.jobType, data.fee, data.status, data.notes, new Date()]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader("Access-Control-Allow-Origin", "*");
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader("Access-Control-Allow-Origin", "*");
  } finally {
    lock.releaseLock();
  }
}
